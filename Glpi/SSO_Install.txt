Plugin install steps/latest fixed 

GLPI 11

# Enter GLPI container
docker exec -it glpi bash

# Remove old plugin if exists
rm -rf /var/www/glpi/plugins/singlesignon

# Install tools
apt-get update && apt-get install -y wget unzip

# Go to plugins folder
cd /var/www/glpi/plugins

# Download LATEST master branch (has fixes for GLPI 11.0.4)
wget https://github.com/edgardmessias/glpi-singlesignon/archive/refs/heads/master.zip

# Extract
unzip master.zip

# Rename to correct folder name
mv glpi-singlesignon-master singlesignon

# Fix permissions
chown -R www-data:www-data singlesignon
chmod -R 755 singlesignon

# Cleanup
rm -f master.zip

# Exit
exit
